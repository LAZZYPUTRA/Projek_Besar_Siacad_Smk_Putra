document.addEventListener('DOMContentLoaded', () => {
    // Handling Dropdown User Nav
    const userMenu = document.querySelector('.user-menu');
    if (userMenu) {
        userMenu.addEventListener('click', (e) => {
            userMenu.classList.toggle('is-open');
            e.stopPropagation();
        });
        document.addEventListener('click', () => userMenu.classList.remove('is-open'));
    }

    // Handling Native Modal (Open)
    document.querySelectorAll('[data-modal-target]').forEach(trigger => {
        trigger.addEventListener('click', (e) => {
            e.preventDefault();
            const targetId = trigger.getAttribute('data-modal-target');
            const modal = document.querySelector(targetId);
            if (modal) {
                modal.classList.add('is-active');
                document.body.style.overflow = 'hidden';
            }
        });
    });

    // Handling Native Modal (Close)
    document.querySelectorAll('[data-modal-close]').forEach(closeBtn => {
        closeBtn.addEventListener('click', () => {
            const modal = closeBtn.closest('.modal-overlay');
            if (modal) {
                modal.classList.remove('is-active');
                document.body.style.overflow = '';
            }
        });
    });

    // Close Modal when clicking outer backdrop
    document.querySelectorAll('.modal-overlay').forEach(modal => {
        modal.addEventListener('click', (e) => {
            if (e.target === modal) {
                modal.classList.remove('is-active');
                document.body.style.overflow = '';
            }
        });
    });
});

/**
 * Membersihkan input form dari karakter berbahaya (Anti-XSS & Trim)
 */
function sanitize($data) {
    if (is_null($data)) return '';
    $data = trim($data);
    $data = stripslashes($data);
    return htmlspecialchars($data, ENT_QUOTES, 'UTF-8');
}

/**
 * Menyiapkan pesan notifikasi temporary (flash message)
 * $tipe: 'success', 'danger', 'warning', 'info'
 */
function set_flash_message($tipe, $pesan) {
    $_SESSION['flash_message'] = [
        'tipe' => $tipe,
        'pesan' => $pesan
    ];
}

/**
 * Menampilkan pesan notifikasi dalam bentuk Alert Bootstrap 5
 */
function display_flash_message() {
    if (isset($_SESSION['flash_message'])) {
        $tipe = $_SESSION['flash_message']['tipe'];
        $pesan = $_SESSION['flash_message']['pesan'];
        
        echo '<div class="alert alert-' . $tipe . ' alert-dismissible fade show shadow-sm" role="alert">';
        echo '<i class="bi bi-info-circle-fill me-2"></i>' . $pesan;
        echo '<button type="button" class="btn-close" data-bs-dismiss="alert" aria-label="Close"></button>';
        echo '</div>';
        
        unset($_SESSION['flash_message']);
    }
}

            e.preventDefault();
            const targetSelector = trigger.getAttribute('data-modal-target');
            const modal = document.querySelector(targetSelector);

                modal.classList.add('is-active');
                modal.style.display = 'flex';

        const closer = e.target.closest('[data-modal-close]');
        if (closer) {
            e.preventDefault();
            const modal = closer.closest('.modal-overlay');
            if (modal) {
                modal.classList.remove('is-active');
                modal.style.display = 'none';
            }
        }

        // Ketika area gelap di luar kotak modal diklik
        if (e.target.classList.contains('modal-overlay')) {

/**
 * Memastikan role pengguna sesuai dengan halaman yang diakses
 * Contoh penggunaan: check_role(['admin']);
 */
function check_role($allowed_roles = []) {
    check_login();
    
    if (!in_array($_SESSION['role'], $allowed_roles)) {
        $role_sekarang = $_SESSION['role'];
        set_flash_message('danger', 'Anda tidak memiliki hak akses untuk membuka halaman tersebut.');
        header("Location: " . BASE_URL . $role_sekarang . "/dashboard_" . $role_sekarang . ".php");
        exit();
    }
}
