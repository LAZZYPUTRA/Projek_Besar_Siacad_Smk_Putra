<?php

    $role = $_SESSION['role'];
    header("Location: " . BASE_URL . $role . "/dashboard_" . $role . ".php");
    exit();
}

$error = '';

// Memproses saat form dikirim (POST)
if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $username = trim($_POST['username']);
    $password = trim($_POST['password']);

    if (empty($username) || empty($password)) {
        $error = 'Username dan Password wajib diisi!';
    } else {
        // Menggunakan Prepared Statement untuk keamanan dari SQL Injection
        $stmt = mysqli_prepare($conn, "SELECT id, username, password, role FROM users WHERE username = ?");
        mysqli_stmt_bind_param($stmt, "s", $username);
        mysqli_stmt_execute($stmt);
        $result = mysqli_stmt_get_result($stmt);

        if ($user = mysqli_fetch_assoc($result)) {
            // Verifikasi Password Hash
            if (password_verify($password, $user['password'])) {
                // Set Session Pengguna
                $_SESSION['user_id']  = $user['id'];
                $_SESSION['username'] = $user['username'];
                $_SESSION['role']     = $user['role'];

                set_flash_message('success', 'Selamat datang kembali, ' . htmlspecialchars($user['username']) . '!');

                // Redirect sesuai role
                header("Location: " . BASE_URL . $user['role'] . "/dashboard_" . $user['role'] . ".php");
                exit();
            } else {
                $error = 'Password yang Anda masukkan salah!';
            }
        } else {
            $error = 'Username tidak ditemukan!';
        }
        mysqli_stmt_close($stmt);
    }
}

/**
 * Membersihkan input form dari karakter berbahaya (Anti-XSS & Trim)
 */
function sanitize($data) {
    if (is_null($data)) return '';
    $data = trim($data);
    $data = stripslashes($data);
    return htmlspecialchars($data, ENT_QUOTES, 'UTF-8');
}

/**
 * Menyiapkan pesan notifikasi temporary (flash message)
 * $tipe: 'success', 'danger', 'warning', 'info'
 */
function set_flash_message($tipe, $pesan) {
    $_SESSION['flash_message'] = [
        'tipe' => $tipe,
        'pesan' => $pesan
    ];
}

/**
 * Menampilkan pesan notifikasi dalam bentuk Alert Bootstrap 5
 */
function display_flash_message() {
    if (isset($_SESSION['flash_message'])) {
        $tipe = $_SESSION['flash_message']['tipe'];
        $pesan = $_SESSION['flash_message']['pesan'];
        
        echo '<div class="alert alert-' . $tipe . ' alert-dismissible fade show shadow-sm" role="alert">';
        echo '<i class="bi bi-info-circle-fill me-2"></i>' . $pesan;
        echo '<button type="button" class="btn-close" data-bs-dismiss="alert" aria-label="Close"></button>';
        echo '</div>';
        
        unset($_SESSION['flash_message']);
    }
}
?>

/**
 * Memastikan pengguna sudah melakukan login
 */
function check_login() {
    if (!isset($_SESSION['user_id'])) {
        $_SESSION['flash_message'] = [
            'tipe' => 'warning',
            'pesan' => 'Silakan login terlebih dahulu untuk mengakses halaman ini.'
        ];
        header("Location: " . BASE_URL . "auth/form_login.php");
        exit();
    }
}

/**
 * Memastikan role pengguna sesuai dengan halaman yang diakses
 * Contoh penggunaan: check_role(['admin']);
 */
function check_role($allowed_roles = []) {
    check_login();
    
    if (!in_array($_SESSION['role'], $allowed_roles)) {
        $role_sekarang = $_SESSION['role'];
        $_SESSION['flash_message'] = [
            'tipe' => 'danger',
            'pesan' => 'Anda tidak memiliki hak akses untuk membuka halaman tersebut.'
        ];
        header("Location: " . BASE_URL . $role_sekarang . "/dashboard_" . $role_sekarang . ".php");
        exit();
    }
}
